```javascript
import { createContext, useContext, useEffect, useState } from 'react';

const ThemeContext = createContext();

export function ThemeProvider({ children }) {
  const [theme, setTheme] = useState(getDefaultTheme());

  useEffect(() => {
    // Listen for theme updates from main app
    const handleThemeUpdate = (event) => {
      if (event.data.type === 'HIGGSFIELD_THEME_UPDATE') {
        setTheme(event.data.theme);
        localStorage.setItem('higgsfield-theme', JSON.stringify(event.data.theme));
      }
    };

    window.addEventListener('message', handleThemeUpdate);
    return () => window.removeEventListener('message', handleThemeUpdate);
  }, []);

  return (
    <ThemeContext.Provider value={{ theme, setTheme }}>
      {children}
    </ThemeContext.Provider>
  );
}

export function useTheme() {
  return useContext(ThemeContext);
}

  const saved = localStorage.getItem('higgsfield-theme');

    colors: {
      primary: urlParams.get('primary') || '#007bff',
      secondary: urlParams.get('secondary') || '#6c757d',
      accent: urlParams.get('accent') || '#007bff',
    },
    logo: urlParams.get('logo') || '/default-logo.png',
    name: urlParams.get('brand') || 'Higgsfield',
```

```javascript
      path,
      context: {

        ...context
      }
    }, '*');
  };

  return { navigateToMainApp };
}
```

```javascript
// apps/remix-go/src/lib/session.js
class SessionManager {
  constructor() {
    this.sessionId = null;
    this.user = null;
    this.init();
  }

  init() {
    // Try to restore session from main app
    this.requestSessionFromMainApp();

    // Listen for session updates
    window.addEventListener('message', (event) => {
      if (event.data.type === 'SESSION_UPDATE') {
        this.updateSession(event.data.session);
      }
    });
  }

  requestSessionFromMainApp() {
    window.parent.postMessage({
      type: 'REQUEST_SESSION',
      app: 'remix-go'
    }, '*');
  }

  updateSession(sessionData) {
    this.sessionId = sessionData.id;
    this.user = sessionData.user;
    localStorage.setItem('remix-go-session', JSON.stringify(sessionData));

    // Notify app of session change
    window.dispatchEvent(new CustomEvent('session-changed', {
      detail: sessionData
    }));
  }

  isAuthenticated() {
    return !!this.user && !!this.sessionId;
  }

  logout() {
    // Clear local session
    this.sessionId = null;
    this.user = null;
    localStorage.removeItem('remix-go-session');

    // Notify main app to logout
    window.parent.postMessage({
      type: 'LOGOUT_REQUEST',
      from: 'remix-go'
    }, '*');
  }
}

export const sessionManager = new SessionManager();
```

```javascript
// lib/database/projects.js - Shared project data operations
import { supabase } from './client';

export class ProjectManager {
  // Create project accessible from both apps
  static async createProject(projectData, userId) {
    const { data, error } = await supabase
      .from('projects')
      .insert({
        ...projectData,
        user_id: userId,
        created_at: new Date().toISOString(),
        updated_at: new Date().toISOString(),
        app_source: 'remix-go', // Track which app created it
      })
      .select()
      .single();

    if (error) throw error;
    return data;
  }

  // Get projects with cross-app visibility
  static async getUserProjects(userId, options = {}) {
    const { app_filter, limit = 50, offset = 0 } = options;

    let query = supabase
      .from('projects')
      .select('*')
      .eq('user_id', userId)
      .order('updated_at', { ascending: false })
      .range(offset, offset + limit - 1);

    if (app_filter) {
      query = query.eq('app_source', app_filter);
    }

    const { data, error } = await query;
    if (error) throw error;
    return data;
  }

  // Real-time subscription for project updates
  static subscribeToProjects(userId, callback) {
    return supabase
      .channel('projects')
      .on('postgres_changes', {
        event: '*',
        schema: 'public',
        table: 'projects',
        filter: `user_id=eq.${userId}`
      }, callback)
      .subscribe();
  }

  // Sync project between apps
  static async syncProjectToMainApp(projectId) {
    // Notify main app of project updates
    window.parent.postMessage({
      type: 'PROJECT_SYNC',
      projectId,
      action: 'updated'
    }, '*');
  }
}
```

```javascript
// lib/database/media.js - Shared media asset management
import { supabase } from './client';

export class MediaManager {
  constructor(bucketName = 'media-assets') {
    this.bucket = bucketName;
  }

  // Upload file to shared storage
  async uploadFile(file, path, options = {}) {
    const { data, error } = await supabase.storage
      .from(this.bucket)
      .upload(path, file, {
        cacheControl: '3600',
        upsert: false,
        ...options
      });

    if (error) throw error;

    // Create database record
    const { data: record, error: dbError } = await supabase
      .from('media_assets')
      .insert({
        filename: file.name,
        path: data.path,
        size: file.size,
        mime_type: file.type,
        bucket: this.bucket,
        uploaded_by: options.userId,
        created_at: new Date().toISOString(),
      })
      .select()
      .single();

    if (dbError) throw dbError;
    return { ...data, record };
  }

  // Get public URL for media asset
  getPublicUrl(path) {
    const { data } = supabase.storage
      .from(this.bucket)
      .getPublicUrl(path);

    return data.publicUrl;
  }

  // Download file with progress tracking
  async downloadFile(path, onProgress) {
    const { data, error } = await supabase.storage
      .from(this.bucket)
      .download(path);

    if (error) throw error;
    return data;
  }

  // Delete file and database record
  async deleteFile(path) {
    // Delete from storage
    const { error: storageError } = await supabase.storage
      .from(this.bucket)
      .remove([path]);

    if (storageError) throw storageError;

    // Delete database record
    const { error: dbError } = await supabase
      .from('media_assets')
      .delete()
      .eq('path', path);

    if (dbError) throw dbError;
  }
```

**Real-Time Synchronization:**
```javascript
// lib/sync/realtime-sync.js
import { supabase } from '../database/client';

export class RealtimeSync {
  constructor() {
    this.subscriptions = new Map();
    this.syncQueue = [];
  }

  // Subscribe to real-time updates
  subscribe(table, filter, callback) {
    const channel = supabase.channel(`${table}-sync`)
      .on('postgres_changes', {
        event: '*',
        schema: 'public',
        table,
        filter
      }, (payload) => {
        // Queue sync operation
        this.syncQueue.push({
          type: payload.eventType,
          table,
          data: payload.new || payload.old,
          callback
        });

        // Process queue
        this.processSyncQueue();
      })
      .subscribe();

    this.subscriptions.set(`${table}-${filter}`, channel);
    return channel;
  }

  // Process queued sync operations
  async processSyncQueue() {
    while (this.syncQueue.length > 0) {
      const operation = this.syncQueue.shift();

      try {
        await this.processSyncOperation(operation);
      } catch (error) {
        console.error('Sync operation failed:', error);
        // Retry logic or error handling
      }
    }
  }

  // Process individual sync operation
  async processSyncOperation(operation) {
    const { type, table, data, callback } = operation;

    // Notify other app of changes
    window.parent.postMessage({
      type: 'DATA_SYNC',
      table,
      operation: type,
      data
    }, '*');

    // Execute callback if provided
    if (callback) {
      callback(operation);
    }
  }

  // Cleanup subscriptions
  unsubscribe(table, filter) {
    const key = `${table}-${filter}`;
    const channel = this.subscriptions.get(key);
    if (channel) {
      supabase.removeChannel(channel);
      this.subscriptions.delete(key);
    }
  }

  // Force sync data between apps
  async forceSync(table, userId) {
    const { data, error } = await supabase
      .from(table)
      .select('*')
      .eq('user_id', userId);

    if (error) throw error;

    // Send all data to other app
    window.parent.postMessage({
      type: 'FORCE_SYNC',
      table,
      data
    }, '*');
  }
}

export const realtimeSync = new RealtimeSync();
```

```javascript
  // Create git tag
  execSync(`git add .`);
  execSync(`git commit -m "Release ${version}"`);
  execSync(`git tag v${version}`);

  // Push changes and tag
  execSync(`git push origin main`);
  execSync(`git push origin v${version}`);

  // Wait for staging tests
  console.log('Running staging tests...');
  execSync('npm run test:staging');

  // Deploy to production
  console.log('Deploying to production...');
  execSync('npm run deploy:production');
```

```javascript
// When user visits the page, personalization runs automatically
class VideoPersonalizationEngine {
  constructor(userData, videoData, template) {
    this.user = userData;
    this.video = videoData;
    this.template = template;
    this.appliedRules = [];
  }

  applyPersonalization() {
    // Evaluate all personalization rules
    this.template.personalizationRules.forEach(rule => {
      if (this.evaluateRule(rule.condition)) {
        this.applyModifications(rule.modifications);
        this.appliedRules.push(rule.id);

        // Track personalization application
        this.trackPersonalization(rule.id);
      }
    });

    // Apply video-specific personalization
    this.personalizeVideoPlayer();
  }

  evaluateRule(condition) {
    try {
      // Safe evaluation of personalization conditions
      const user = this.user;
      const device = this.detectDevice();
      const time = this.getCurrentTime();

      return eval(condition); // In production, use a safe evaluation library
    } catch (error) {
      console.warn('Personalization rule evaluation failed:', error);
      return false;
    }
  }

  applyModifications(modifications) {
    Object.entries(modifications).forEach(([componentId, changes]) => {
      const component = this.findComponent(componentId);
      if (component) {
        this.updateComponent(component, changes);
      }
    });
  }

  personalizeVideoPlayer() {
    const videoPlayer = this.findComponent('sendspark-video');
    if (!videoPlayer) return;

    // Apply user-specific video settings
    if (this.user.preferences.autoplay === false) {
      videoPlayer.set('autoplay', false);
    }

    if (this.user.device.type === 'mobile') {
      videoPlayer.set('preload', 'none');
      videoPlayer.set('controls', true);
    }

    // Add personalized overlay
    const overlayText = this.generatePersonalizedOverlay();
    videoPlayer.addOverlay(overlayText);
  }

  generatePersonalizedOverlay() {
    if (this.appliedRules.includes('first-time-visitor')) {
      return `👋 Welcome ${this.user.name || 'friend'}!`;
    }

    if (this.appliedRules.includes('enterprise-segment')) {
      return '🏢 Enterprise Solutions Available';
    }

    return this.video.title; // Default to video title
  }

  trackPersonalization(ruleId) {
    // Send analytics data
    if (window.analytics) {
      window.analytics.track('personalization_applied', {
        ruleId,
        userId: this.user.id,
        videoId: this.video.id,
        timestamp: Date.now()
      });
    }
  }
}

  currentUser,
  createdVideo,
  videoLandingTemplate
```

```javascript
// Track video performance and personalization effectiveness
const videoAnalytics = {
  videoMetrics: {
    views: 0,
    watchTime: 0,
    completionRate: 0,
    dropOffPoints: [],
    engagementScore: 0
  },

  personalizationMetrics: {
    rulesApplied: [],
    conversionLift: 0,
    abTestResults: {},
    userSegments: {}
  },

  trackVideoEvent: (eventType, data) => {
    // Track video-specific events
    switch (eventType) {
      case 'play':
        this.videoMetrics.views++;
        this.trackPersonalizationImpact('video_started');
        break;
      case 'complete':
        this.videoMetrics.completionRate =
          (this.videoMetrics.completionRate + 1) / this.videoMetrics.views;
        this.trackConversion('video_completed');
        break;
      case 'cta_click':
        this.trackConversion('cta_clicked', data.buttonType);
        break;
    }
  },

  trackPersonalizationImpact: (event) => {
    // Measure how personalization affects engagement
    const baselineEngagement = 0.65; // Industry average
    const actualEngagement = this.calculateEngagement();

    this.personalizationMetrics.conversionLift =
      ((actualEngagement - baselineEngagement) / baselineEngagement) * 100;
  },

  calculateEngagement: () => {
    return this.videoMetrics.watchTime / this.video.duration;
  }
};
```
